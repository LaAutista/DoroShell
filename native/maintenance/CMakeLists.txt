# The published single-file bootstrap verifies both C++ sources before compiling.
file(READ "${CMAKE_CURRENT_LIST_DIR}/../../bootstrap.sh" download_entry)
foreach(input IN ITEMS native/maintenance/main.cpp native/compatibility/packages.hpp)
    set(file "${CMAKE_CURRENT_LIST_DIR}/../../${input}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${file}")
    file(SHA256 "${file}" digest)
    string(FIND "${download_entry}" "${digest}  ${input}" match)
    if(match EQUAL -1)
        message(FATAL_ERROR "Update bootstrap.sh SHA256 for ${input}")
    endif()
endforeach()
