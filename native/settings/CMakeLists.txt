cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(DoroShellSettings LANGUAGES CXX)
    include(CTest)
endif()
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SETTINGS_GIO REQUIRED IMPORTED_TARGET gio-unix-2.0)
add_library(doro-settings-window STATIC hub.cpp)
target_compile_features(doro-settings-window PUBLIC cxx_std_20)
target_compile_definitions(doro-settings-window PRIVATE QT_NO_KEYWORDS)
target_link_libraries(doro-settings-window PUBLIC Qt6::Widgets PRIVATE PkgConfig::SETTINGS_GIO)
add_executable(doroshell-settings main.cpp)
target_link_libraries(doroshell-settings PRIVATE doro-settings-window)
install(TARGETS doroshell-settings RUNTIME DESTINATION bin)
if(BUILD_TESTING)
    add_executable(doro-settings-test test.cpp)
    target_link_libraries(doro-settings-test PRIVATE doro-settings-window)
    add_test(NAME settings-native-actions COMMAND doro-settings-test)
endif()
