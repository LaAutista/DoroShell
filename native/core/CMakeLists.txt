cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(DoroShellCore LANGUAGES CXX)
    include(CTest)
endif()
find_package(Qt6 REQUIRED COMPONENTS Core)
add_executable(doroshell-core main.cpp)
target_compile_features(doroshell-core PRIVATE cxx_std_20)
target_link_libraries(doroshell-core PRIVATE Qt6::Core)
install(TARGETS doroshell-core RUNTIME DESTINATION bin)
if(BUILD_TESTING)
    set(DOROSHELL_CORE_TEST_RUNTIME "" CACHE PATH "Previously built complete patched native bundle for core integration testing")
    if(DOROSHELL_CORE_TEST_RUNTIME)
        add_executable(doroshell-core-test core_test.cpp)
        target_compile_features(doroshell-core-test PRIVATE cxx_std_20)
        target_link_libraries(doroshell-core-test PRIVATE Qt6::Core)
        add_test(NAME native-core-complete-install-recovery COMMAND doroshell-core-test
                 $<TARGET_FILE:doroshell-core> "${DOROSHELL_CORE_TEST_RUNTIME}" "${DOROSHELL_PRIVATE_XVFB}")
    endif()
endif()
