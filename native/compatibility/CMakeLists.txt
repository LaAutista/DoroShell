cmake_minimum_required(VERSION 3.22)
# Builds alone (cmake -S native/compatibility) and under the top-level add_subdirectory.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(DoroShellCompatibility LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  include(GNUInstallDirs)
  include(CTest)
endif()
find_package(Qt6 REQUIRED COMPONENTS Core)
add_executable(doroshell-check main.cpp)
target_link_libraries(doroshell-check PRIVATE Qt6::Core)
install(TARGETS doroshell-check)
if(BUILD_TESTING)
  add_executable(native-compatibility-test check_test.cpp)
  target_link_libraries(native-compatibility-test PRIVATE Qt6::Core)
  add_test(NAME native-compatibility COMMAND native-compatibility-test $<TARGET_FILE:doroshell-check>)
endif()
