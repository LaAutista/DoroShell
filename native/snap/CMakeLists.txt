cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(DoroShellSnap LANGUAGES CXX)
  include(CTest)
endif()
find_package(Qt6 REQUIRED COMPONENTS Core Network)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SNAP_XCB REQUIRED IMPORTED_TARGET xcb xcb-randr)
add_executable(doroshell-snap main.cpp snap.cpp)
target_compile_features(doroshell-snap PRIVATE cxx_std_20)
target_link_libraries(doroshell-snap PRIVATE Qt6::Core Qt6::Network PkgConfig::SNAP_XCB)
install(TARGETS doroshell-snap RUNTIME DESTINATION bin)

set(DOROSHELL_SNAP_XVFB "" CACHE FILEPATH "Private Xvfb executable for snap integration tests")
if(BUILD_TESTING AND DOROSHELL_SNAP_XVFB)
  pkg_check_modules(SNAP_XTEST REQUIRED IMPORTED_TARGET xcb-xtest)
  add_executable(snap-private-x11-test private_test.cpp)
  target_compile_features(snap-private-x11-test PRIVATE cxx_std_20)
  target_link_libraries(snap-private-x11-test PRIVATE Qt6::Core Qt6::Network PkgConfig::SNAP_XCB PkgConfig::SNAP_XTEST)
  add_test(NAME snap-private-x11 COMMAND snap-private-x11-test "${DOROSHELL_SNAP_XVFB}" $<TARGET_FILE:doroshell-snap>)
  set_tests_properties(snap-private-x11 PROPERTIES TIMEOUT 40)
endif()
