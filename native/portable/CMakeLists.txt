cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(DoroShellPortable LANGUAGES CXX)
    include(CTest)
endif()
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DORO_PORTABLE_JSON REQUIRED IMPORTED_TARGET jansson)
add_library(doroshell-portable-core STATIC portable.cpp store.cpp)
target_compile_features(doroshell-portable-core PUBLIC cxx_std_20)
target_link_libraries(doroshell-portable-core PUBLIC Qt6::Core PkgConfig::DORO_PORTABLE_JSON)
add_executable(doroshell-portable main.cpp)
target_link_libraries(doroshell-portable PRIVATE doroshell-portable-core Qt6::Widgets)
install(TARGETS doroshell-portable RUNTIME DESTINATION bin)
if(BUILD_TESTING)
    add_executable(doroshell-portable-test portable_test.cpp)
    target_link_libraries(doroshell-portable-test PRIVATE doroshell-portable-core)
    add_test(NAME doroshell-portable COMMAND doroshell-portable-test)
endif()
