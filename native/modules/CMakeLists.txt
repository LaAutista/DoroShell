cmake_minimum_required(VERSION 3.22)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    project(DoroShellModules LANGUAGES CXX)
    include(CTest)
    add_subdirectory(../theme theme)
endif()
find_package(Qt6 REQUIRED COMPONENTS Core)
add_executable(doroshell-modules main.cpp options.cpp)
target_compile_features(doroshell-modules PRIVATE cxx_std_20)
target_link_libraries(doroshell-modules PRIVATE doroshell-theme-core)
install(TARGETS doroshell-modules RUNTIME DESTINATION bin)
if(BUILD_TESTING)
    add_executable(native-modules-test test.cpp)
    target_compile_features(native-modules-test PRIVATE cxx_std_20)
    target_link_libraries(native-modules-test PRIVATE Qt6::Core)
    add_test(NAME native-optional-modules COMMAND native-modules-test $<TARGET_FILE:doroshell-modules>)
endif()
