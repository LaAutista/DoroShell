cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(DoroShellNetwork LANGUAGES CXX)
    include(CTest)
endif()
find_package(Qt6 REQUIRED COMPONENTS Core Widgets DBus)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DORO_NM REQUIRED IMPORTED_TARGET libnm>=1.44 libnma gtk+-3.0 gio-unix-2.0)
add_library(doroshell-network-model STATIC model.cpp)
target_compile_features(doroshell-network-model PUBLIC cxx_std_20)
target_compile_definitions(doroshell-network-model PUBLIC QT_NO_KEYWORDS)
target_link_libraries(doroshell-network-model PUBLIC PkgConfig::DORO_NM Qt6::Core)
add_library(doroshell-network-ui STATIC chooser.cpp)
add_executable(doroshell-network main.cpp)
target_link_libraries(doroshell-network PRIVATE doroshell-network-ui)
set_target_properties(doroshell-network-ui PROPERTIES AUTOMOC ON)
target_link_libraries(doroshell-network-ui PUBLIC doroshell-network-model Qt6::Widgets Qt6::DBus)
install(TARGETS doroshell-network RUNTIME DESTINATION bin)
if(BUILD_TESTING)
    add_executable(doroshell-network-test model_test.cpp)
    target_link_libraries(doroshell-network-test PRIVATE doroshell-network-model)
    add_test(NAME doroshell-network-model COMMAND doroshell-network-test)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(doroshell-network-ui-test chooser_test.cpp)
    target_link_libraries(doroshell-network-ui-test PRIVATE doroshell-network-ui Qt6::Test)
    add_test(NAME doroshell-network-private-bus COMMAND doroshell-network-ui-test)
    set_tests_properties(doroshell-network-private-bus PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
endif()
