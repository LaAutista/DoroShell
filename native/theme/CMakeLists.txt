cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(DoroShellTheme LANGUAGES CXX)
    include(CTest)
endif()
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
find_package(EXPAT REQUIRED)
add_library(doroshell-theme-core STATIC palette.cpp edits.cpp transaction.cpp)
target_compile_features(doroshell-theme-core PUBLIC cxx_std_20)
target_link_libraries(doroshell-theme-core PUBLIC Qt6::Core Qt6::Gui EXPAT::EXPAT)
set(templates start-theme-glass.qss panel-theme-glass.qss scheme.colors kate.json themerc)
list(TRANSFORM templates PREPEND "templates/")
qt_add_resources(doroshell-theme-core palette PREFIX /theme BASE templates FILES ${templates})
add_executable(doroshell-theme main.cpp)
target_link_libraries(doroshell-theme PRIVATE doroshell-theme-core)
install(TARGETS doroshell-theme RUNTIME DESTINATION bin)
if(BUILD_TESTING)
    add_executable(doroshell-theme-test theme_test.cpp)
    target_link_libraries(doroshell-theme-test PRIVATE doroshell-theme-core)
    add_test(NAME doroshell-theme COMMAND doroshell-theme-test)
endif()
