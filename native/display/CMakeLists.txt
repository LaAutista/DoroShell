cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(DoroShellDisplay LANGUAGES CXX)
  include(CTest)
endif()
find_package(Qt6 REQUIRED COMPONENTS Core)
add_executable(doroshell-display backend.cpp state.cpp)
target_compile_features(doroshell-display PRIVATE cxx_std_20)
target_link_libraries(doroshell-display PRIVATE Qt6::Core)
install(TARGETS doroshell-display RUNTIME DESTINATION bin)

if(BUILD_TESTING)
  add_executable(display-state-test state_test.cpp state.cpp)
  target_compile_features(display-state-test PRIVATE cxx_std_20)
  target_link_libraries(display-state-test PRIVATE Qt6::Core)
  add_test(NAME native-display-state COMMAND display-state-test)
  set(DOROSHELL_DISPLAY_XVFB "" CACHE FILEPATH "Private programmable Xvfb for display guard tests")
  if(DOROSHELL_DISPLAY_XVFB)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(DISPLAY_TEST_XCB REQUIRED IMPORTED_TARGET xcb)
    add_executable(display-private-x11-test private_test.cpp)
    target_compile_features(display-private-x11-test PRIVATE cxx_std_20)
    target_link_libraries(display-private-x11-test PRIVATE Qt6::Core PkgConfig::DISPLAY_TEST_XCB)
    add_test(NAME display-private-x11 COMMAND display-private-x11-test "${DOROSHELL_DISPLAY_XVFB}" $<TARGET_FILE:doroshell-display>)
    set_tests_properties(display-private-x11 PROPERTIES TIMEOUT 60)
  endif()
endif()

set(DOROSHELL_PATCHED_DISPLAY_INCLUDE "" CACHE PATH "Prepared native PCManFM pcmanfm/ directory for widget integration testing")
if(BUILD_TESTING AND DOROSHELL_PATCHED_DISPLAY_INCLUDE)
  find_package(Qt6 REQUIRED COMPONENTS Widgets)
  add_executable(display-widget-test widget_test.cpp)
  target_compile_features(display-widget-test PRIVATE cxx_std_20)
  target_include_directories(display-widget-test PRIVATE "${DOROSHELL_PATCHED_DISPLAY_INCLUDE}")
  target_link_libraries(display-widget-test PRIVATE Qt6::Widgets)
  add_test(NAME display-native-widget-launch COMMAND display-widget-test $<TARGET_FILE:doroshell-display>)
endif()
