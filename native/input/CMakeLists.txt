cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(DoroShellInput LANGUAGES CXX)
    include(CTest)
endif()
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DORO_INPUT REQUIRED IMPORTED_TARGET xcb xcb-xinput libudev)
add_library(doroshell-input-core STATIC preferences.cpp xinput.cpp diagnostic.cpp)
target_compile_features(doroshell-input-core PUBLIC cxx_std_20)
target_link_libraries(doroshell-input-core PUBLIC Qt6::Core PkgConfig::DORO_INPUT)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
pkg_check_modules(DORO_INPUT_GIO REQUIRED IMPORTED_TARGET gio-unix-2.0)
add_executable(doroshell-input-overrides overrides.cpp)
target_compile_definitions(doroshell-input-overrides PRIVATE QT_NO_KEYWORDS)
target_link_libraries(doroshell-input-overrides PRIVATE doroshell-input-core Qt6::Widgets PkgConfig::DORO_INPUT_GIO)
add_executable(doroshell-input-preferences main.cpp)
target_link_libraries(doroshell-input-preferences PRIVATE doroshell-input-core)
install(TARGETS doroshell-input-overrides doroshell-input-preferences RUNTIME DESTINATION bin)
if(BUILD_TESTING)
    add_executable(doroshell-input-test input_test.cpp)
    target_link_libraries(doroshell-input-test PRIVATE doroshell-input-core)
    add_test(NAME doroshell-input-preferences COMMAND doroshell-input-test)
    add_executable(doroshell-input-x11-test x11_test.cpp)
    target_link_libraries(doroshell-input-x11-test PRIVATE doroshell-input-core)
    add_dependencies(doroshell-input-x11-test doroshell-input-preferences)
    add_test(NAME doroshell-input-private-x11 COMMAND doroshell-input-x11-test)
    set_tests_properties(doroshell-input-private-x11 PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 25)
    if(DOROSHELL_PRIVATE_XVFB)
        set_tests_properties(doroshell-input-private-x11 PROPERTIES ENVIRONMENT "DOROSHELL_TEST_XVFB=${DOROSHELL_PRIVATE_XVFB}")
    endif()
endif()
add_executable(doroshell-config-input launch_main.cpp launch.cpp)
target_compile_features(doroshell-config-input PRIVATE cxx_std_20)
target_link_libraries(doroshell-config-input PRIVATE Qt6::Core)
install(TARGETS doroshell-config-input RUNTIME DESTINATION bin)
if(BUILD_TESTING)
    add_executable(doroshell-input-launch-test launch_test.cpp launch.cpp)
    target_compile_features(doroshell-input-launch-test PRIVATE cxx_std_20)
    target_link_libraries(doroshell-input-launch-test PRIVATE Qt6::Widgets)
    add_test(NAME native-input-launch-repair COMMAND doroshell-input-launch-test $<TARGET_FILE:doroshell-config-input>)
endif()
