cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(DoroShellPanelProfile LANGUAGES CXX)
  include(CTest)
endif()
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PANEL_PROFILE_GIO REQUIRED IMPORTED_TARGET gio-unix-2.0)
add_executable(doroshell-panel-profile main.cpp profile.cpp)
target_compile_features(doroshell-panel-profile PRIVATE cxx_std_20)
target_link_libraries(doroshell-panel-profile PRIVATE Qt6::Core PkgConfig::PANEL_PROFILE_GIO)
install(TARGETS doroshell-panel-profile RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  add_executable(panel-profile-test profile_test.cpp)
  target_compile_features(panel-profile-test PRIVATE cxx_std_20)
  target_link_libraries(panel-profile-test PRIVATE Qt6::Core)
  add_test(NAME native-panel-profile-transaction COMMAND panel-profile-test $<TARGET_FILE:doroshell-panel-profile>)
endif()
set(DOROSHELL_PANEL_TEST_BINARY "" CACHE FILEPATH "Patched private lxqt-panel for the staged profile X11 test")
if(BUILD_TESTING AND DOROSHELL_PRIVATE_XVFB AND DOROSHELL_PANEL_TEST_BINARY)
  pkg_check_modules(PROFILE_XCB REQUIRED IMPORTED_TARGET xcb)
  add_executable(panel-profile-private-test private_test.cpp)
  target_compile_features(panel-profile-private-test PRIVATE cxx_std_20)
  target_link_libraries(panel-profile-private-test PRIVATE Qt6::Core PkgConfig::PROFILE_XCB)
  add_test(NAME native-panel-staged-x11 COMMAND panel-profile-private-test "${DOROSHELL_PRIVATE_XVFB}"
           $<TARGET_FILE:doroshell-panel-profile> "${DOROSHELL_PANEL_TEST_BINARY}")
  set_tests_properties(native-panel-staged-x11 PROPERTIES TIMEOUT 45)
endif()
