cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(DoroShellCursor LANGUAGES CXX)
    include(CTest)
endif()
find_package(Qt6 REQUIRED COMPONENTS Gui)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CURSOR_XCURSOR REQUIRED IMPORTED_TARGET xcursor)
find_package(ZLIB REQUIRED)
add_library(doro-cursor-core STATIC animation.cpp cursor.cpp transaction.cpp)
target_compile_features(doro-cursor-core PUBLIC cxx_std_20)
target_link_libraries(doro-cursor-core PUBLIC Qt6::Gui PkgConfig::CURSOR_XCURSOR ZLIB::ZLIB)
add_executable(doroshell-cursor main.cpp)
target_link_libraries(doroshell-cursor PRIVATE doro-cursor-core)
set(cursor_assets "${CMAKE_CURRENT_SOURCE_DIR}/../../assets/cursors")
if(EXISTS "${cursor_assets}/spin-doro.gif" AND EXISTS "${cursor_assets}/spin-doro-transparent.apng")
    qt_add_resources(doroshell-cursor cursor_art PREFIX "/doro/cursors" BASE "${cursor_assets}"
        FILES "${cursor_assets}/spin-doro.gif" "${cursor_assets}/spin-doro-transparent.apng")
endif()
install(TARGETS doroshell-cursor RUNTIME DESTINATION bin)
if(BUILD_TESTING)
    add_executable(doro-cursor-test test.cpp)
    target_link_libraries(doro-cursor-test PRIVATE doro-cursor-core)
    add_test(NAME cursor-native-install-restore COMMAND doro-cursor-test $<TARGET_FILE:doroshell-cursor>)
endif()
