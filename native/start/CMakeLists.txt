cmake_minimum_required(VERSION 3.22)
project(DoroShellStart LANGUAGES CXX)
include(CTest)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network DBus)
find_package(PkgConfig REQUIRED)
pkg_check_modules(START_GIO REQUIRED IMPORTED_TARGET gio-unix-2.0)
pkg_check_modules(START_XCB REQUIRED IMPORTED_TARGET xcb)
add_library(doro-start-model STATIC model.cpp browse.cpp)
target_link_libraries(doro-start-model PUBLIC Qt6::Widgets PkgConfig::START_GIO)
target_compile_definitions(doro-start-model PUBLIC QT_NO_KEYWORDS)
target_compile_features(doro-start-model PUBLIC cxx_std_20)
add_executable(doroshell-start main.cpp focus.cpp preferences.cpp)
target_link_libraries(doroshell-start PRIVATE doro-start-model Qt6::Network Qt6::DBus PkgConfig::START_XCB)
install(TARGETS doroshell-start RUNTIME DESTINATION bin)
if(BUILD_TESTING)
    add_executable(doro-start-test test.cpp preferences.cpp)
    target_link_libraries(doro-start-test PRIVATE doro-start-model)
    add_test(NAME start-catalog-and-settings COMMAND doro-start-test)
endif()
set(start_art "${CMAKE_CURRENT_SOURCE_DIR}/../../assets/doro/portrait.png")
if(NOT EXISTS "${start_art}")
    set(start_art "${CMAKE_CURRENT_SOURCE_DIR}/../../menu/start.svg")
endif()
get_filename_component(start_art_name "${start_art}" NAME)
set_source_files_properties("${start_art}" PROPERTIES QT_RESOURCE_ALIAS "${start_art_name}")
qt_add_resources(doroshell-start start_artwork PREFIX "/doro" FILES "${start_art}")
if(BUILD_TESTING)
    if(DOROSHELL_PRIVATE_XVFB)
        set(DOROSHELL_START_XVFB "${DOROSHELL_PRIVATE_XVFB}")
    else()
        find_program(DOROSHELL_START_XVFB Xvfb)
    endif()
    if(DOROSHELL_START_XVFB)
        pkg_check_modules(START_XTEST REQUIRED IMPORTED_TARGET xcb-xtest xcb-keysyms)
        add_executable(doro-start-x11-test x11-test.cpp)
        target_link_libraries(doro-start-x11-test PRIVATE Qt6::Widgets PkgConfig::START_XCB PkgConfig::START_XTEST)
        target_compile_features(doro-start-x11-test PRIVATE cxx_std_20)
        add_test(NAME start-x11-focus-and-toggle COMMAND doro-start-x11-test "${DOROSHELL_START_XVFB}" $<TARGET_FILE:doroshell-start>)
    endif()
endif()
qt_add_resources(doroshell-start start_qss PREFIX "/doro" FILES start.qss)
