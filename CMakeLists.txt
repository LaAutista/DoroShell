cmake_minimum_required(VERSION 3.22)
project(DoroShell LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CTest)
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus Network)
set(DOROSHELL_PRIVATE_XVFB "" CACHE FILEPATH "Private Xvfb executable for native integration tests")
add_subdirectory(native/clipboard)
add_subdirectory(native/capture)
if(DOROSHELL_PRIVATE_XVFB)
    set(DOROSHELL_SNAP_XVFB "${DOROSHELL_PRIVATE_XVFB}")
    set(DOROSHELL_DISPLAY_XVFB "${DOROSHELL_PRIVATE_XVFB}")
endif()
add_subdirectory(native/snap)
add_subdirectory(native/start)
add_subdirectory(native/network)
add_subdirectory(native/display)
add_subdirectory(native/input)
option(DOROSHELL_BUILD_LXQT_PANEL "Enable the isolated patched LXQt panel build target" OFF)
if(DOROSHELL_BUILD_LXQT_PANEL)
    add_subdirectory(native/upstream)
endif()
add_subdirectory(native/compatibility)
add_subdirectory(native/setup)
add_subdirectory(native/core)
add_subdirectory(native/migration)
add_subdirectory(native/maintenance)
add_subdirectory(native/desktop)
add_subdirectory(native/settings)
add_subdirectory(native/theme)
add_subdirectory(native/modules)
add_subdirectory(native/cursor)
add_subdirectory(native/portable)
add_subdirectory(native/panel-profile)
install(FILES menu/start.svg DESTINATION "${CMAKE_INSTALL_DATADIR}/doroshell")
install(FILES assets/doro/start.png DESTINATION "${CMAKE_INSTALL_DATADIR}/doroshell" OPTIONAL)

if(DOROSHELL_BUILD_LXQT_PANEL)
    set(bundle_apps doroshell-start doroshell-screenshot doroshell-snap doroshell-settings
        doroshell-network doroshell-clipboard doroshell-input-overrides doroshell-input-preferences
        doroshell-display doroshell-desktop doroshell-setup doroshell-panel-profile doroshell-theme
        doroshell-config-input doroshell-cursor doroshell-portable doroshell-check doroshell-core doroshell-modules doroshell-migrate doroshell-route doroshell-maintenance doroshell-wallpaper)
    set(bundle_bin "${CMAKE_BINARY_DIR}/native/upstream/runtime/bin")
    add_custom_target(doroshell-bundle DEPENDS doroshell-screengrab-build doroshell-panel doroshell-desktop-build doroshell-input-build ${bundle_apps})
    add_custom_command(TARGET doroshell-bundle POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E make_directory "${bundle_bin}/../share/doroshell"
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_SOURCE_DIR}/menu/start.svg"
            "${bundle_bin}/../share/doroshell/start.svg")
    if(EXISTS "${CMAKE_SOURCE_DIR}/assets/doro/start.png")
        add_custom_command(TARGET doroshell-bundle POST_BUILD COMMAND "${CMAKE_COMMAND}" -E
            copy_if_different "${CMAKE_SOURCE_DIR}/assets/doro/start.png" "${bundle_bin}/../share/doroshell/start.png")
    endif()
    foreach(app IN LISTS bundle_apps)
        add_custom_command(TARGET doroshell-bundle POST_BUILD COMMAND "${CMAKE_COMMAND}" -E
            copy_if_different "$<TARGET_FILE:${app}>" "${bundle_bin}/$<TARGET_FILE_NAME:${app}>")
    endforeach()
endif()
if(BUILD_TESTING AND DOROSHELL_BUILD_LXQT_PANEL AND DOROSHELL_PRIVATE_XVFB)
    add_test(NAME capture-screengrab-import-x11 COMMAND capture-import-test
        "${DOROSHELL_PRIVATE_XVFB}" "${bundle_bin}/doroshell-screengrab-import")
endif()
if(BUILD_TESTING AND DOROSHELL_BUILD_LXQT_PANEL AND DOROSHELL_PRIVATE_XVFB)
    add_executable(doro-session-login-test native/testing/session_login.cpp)
    target_link_libraries(doro-session-login-test PRIVATE Qt6::Core)
    add_test(NAME native-lxqt-session-startup COMMAND doro-session-login-test "${DOROSHELL_PRIVATE_XVFB}" "${bundle_bin}")
endif()
